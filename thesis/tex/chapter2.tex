\section{مبانی نظری}
\label{sec:ch2_foundations}

\subsection{مسئلهٔ \lr{MDE}}
تخمین عمق تک‌دوربینه (\lr{MDE}) به معنای پیش‌بینی عمق هر پیکسل از یک تصویر رنگی $I\in\mathbb{R}^{H\times W\times3}$ است. مدل یادگیری را می‌توان به‌صورت
\begin{equation}
 D=f_{\theta}(I),
\end{equation}
نوشت که $D\in\mathbb{R}^{H\times W}$ نگاشت عمق و $\theta$ پارامترهای مدل است. این نگاشت از تصویر به صحنهٔ سه‌بعدی یکتا نیست: تغییر هم‌زمان اندازهٔ صحنه، فاصلهٔ کانونی و عمق می‌تواند تصویر یکسانی بسازد. بنابراین، مسئله ذاتاً بدحال است و مدل باید ابهام را با داده، فرض‌های صحنه یا قیدهای هندسی کاهش دهد~\cite{HartleyZisserman2004,Eigen2014Depth}.

در عمق نسبی، تنها ترتیب یا نسبت عمق‌ها اهمیت دارد. اگر $\hat D$ پیش‌بینی مدل باشد، دو خروجی $\hat D$ و $s\hat D+t$ (برای $s>0$) از دید معیارهای تغییرناپذیر نسبت به مقیاس و انتقال معادل‌اند. در عمق متریک، خروجی باید در واحد فیزیکی، معمولاً متر، تفسیر شود؛ ازاین‌رو تغییر مقیاس مجاز نیست و خطاهایی مانند
\begin{equation}
 \mathrm{AbsRel}=\frac{1}{N}\sum_i\frac{|D_i-\hat D_i|}{D_i},\qquad
 \mathrm{RMSE}=\sqrt{\frac{1}{N}\sum_i(D_i-\hat D_i)^2}
\end{equation}
به مقیاس مطلق حساس‌اند؛ در اینجا خطای نسبی مطلق (\lr{AbsRel})\enfootnote{Absolute Relative Error} و ریشهٔ میانگین مربعات خطا (\lr{RMSE})\enfootnote{Root Mean Squared Error} به‌کار می‌روند. این تمایز در ادامه برای جدا کردن دستاوردهای مدل‌های نسبی از مدل‌های متریک ضروری است.

\subsection{مدل سوراخ‌سوزنی و ماتریس کالیبراسیون دوربین}
رابطهٔ تصویربرداری یک نقطهٔ سه‌بعدی $\mathbf X=(X,Y,Z)^\mathsf T$ را با مدل سوراخ‌سوزنی می‌توان چنین نوشت:
\begin{equation}
 s\begin{bmatrix}u\\v\\1\end{bmatrix}=\calibrationmatrix{}\,[R\mid\mathbf t]
 \begin{bmatrix}X\\Y\\Z\\1\end{bmatrix},\qquad
 \calibrationmatrix{}=\begin{bmatrix}f_x&\gamma&c_x\\0&f_y&c_y\\0&0&1\end{bmatrix}.
\end{equation}
در این رابطه $f_x,f_y$ فاصله‌های کانونی بر حسب پیکسل، $(c_x,c_y)$ نقطهٔ اصلی و $\gamma$ کجی محورهای تصویر است. برای دوربین مستطیلی معمولاً $\gamma=0$. پارامترهای درونی ماتریس کالیبراسیون دوربین نگاشت پیکسل و شعاع دید را تعیین می‌کنند؛ پارامترهای بیرونی $(R,\mathbf t)$ وضعیت دوربین نسبت به صحنه را تعیین می‌کنند. در نگاشت یک نقطهٔ تصویری به پرتو سه‌بعدی، عبارت $\calibrationmatrix{}^{-1}[u,v,1]^\mathsf T$ نشان می‌دهد که یک پیکسل بدون دانستن ماتریس کالیبراسیون دوربین تنها جهت تقریبی را مشخص می‌کند، نه مقیاس عمق را.

در خودنظارتی ویدئویی، اگر $\pi$ عمل فرافکنی باشد، بازسازی دیدگاه به‌صورت
\begin{equation}
 \hat I_{t+1}(p)=I_t\left(\pi\left(\calibrationmatrix{}T_{t\rightarrow t+1}D_{t+1}(p)\calibrationmatrix{}^{-1}\tilde p\right)\right)
\end{equation}
انجام می‌شود. این معادله نشان می‌دهد که ماتریس کالیبراسیون دوربین ممکن است در تابع هزینهٔ بازسازی حضور داشته باشد، بی‌آنکه شبکهٔ عمق آن را به‌عنوان ورودی دریافت کند. این دو حالت در این فصل از یکدیگر تفکیک می‌شوند.

\subsection{پارادایم‌های نظارتی و تابع‌های هزینه}
در روش نظارت‌شده، نقشهٔ مرجع $D^{\mathrm{gt}}$ مستقیماً به مدل سیگنال می‌دهد. در روش خودنظارتی استریو، زوج چپ--راست و خط پایه $b$ برای بازسازی تصویر به‌کار می‌رود و رابطهٔ ایده‌آل $Z=fb/d$ مقیاس را به تنظیم استریو وابسته می‌کند. در روش خودنظارتی ویدئویی، شبکهٔ عمق و شبکهٔ ژست با کمینه‌کردن خطای بازسازی فریم‌های زمانی آموزش می‌بینند. در آموزش نیمه‌نظارتی، عمق حسگر و قید سازگاری استریو هم‌زمان استفاده می‌شوند. این تفاوت‌ها بعداً در مقایسهٔ مقالات تاریخی به‌کار می‌آیند.

برای خروجی نسبی، یک هزینهٔ رایج در فضای معکوس عمق، خطای تغییرناپذیر نسبت به مقیاس و انتقال است. برای خروجی متریک، تابع هزینهٔ لگاریتمی تغییرناپذیر نسبت به مقیاس (\lr{SiLog})\enfootnote{Scale-Invariant Logarithmic Loss} به‌صورت
\begin{equation}
 L_{\mathrm{SiLog}}=\frac1N\sum_i g_i^2-\frac{\lambda}{N^2}\left(\sum_i g_i\right)^2,\qquad g_i=\log\hat D_i-\log D_i
\end{equation}
تعریف می‌شود. انتخاب فضای هدف و تابع هزینه باید با معنای خروجی سر مدل سازگار باشد؛ در غیر این صورت، حتی کاهش ظاهری هزینه نیز می‌تواند با عمق صحیح ناسازگار باشد.

\subsection{بازنمایی‌های بصری: ترنسفورمر بینایی (\lr{ViT})، \lr{DINOv2} و ترنسفورمر پیش‌بینی چگال (\lr{DPT})}
ترنسفورمر بینایی (\lr{ViT})\enfootnote{Vision Transformer} تصویر را به قطعه‌های تصویری تبدیل و هر قطعه را به یک توکن نگاشت می‌کند. خودتوجهی چندسری وابستگی‌های دوربرد میان توکن‌ها را مدل می‌کند~\cite{Dosovitskiy2020ViT}. \lr{DINOv2} با یادگیری خودنظارتی روی مجموعهٔ بزرگی از تصاویر، ویژگی‌هایی مقاوم در برابر تغییر دامنه و قابل استفاده برای وظایف چگال تولید می‌کند؛ نکتهٔ مهم برای این پژوهش آن است که این ویژگی‌ها عمدتاً محتوای بصری را کد می‌کنند و پارامترهای دوربین را به‌صورت صریح دریافت نمی‌کنند~\cite{Oquab2023DINOv2}.

ترنسفورمر پیش‌بینی چگال (\lr{DPT})\enfootnote{Dense Prediction Transformer} چند خروجی میانی انکودر را به نگاشت‌های مکانی با وضوح‌های مختلف تبدیل و با یک دیکودر کانولوشنی تدریجی ترکیب می‌کند. در نتیجه، هم میدان دید سراسری ترنسفورمر و هم مرزهای مکانی لازم برای پیش‌بینی چگال حفظ می‌شوند~\cite{Ranftl2021DPT}. این ترکیب، هستهٔ معماری بسیاری از مدل‌های جدید، از جمله \lr{Depth Anything V2}، است.

\section{مرور ادبیات گذشته}
\label{sec:ch2_literature}

\subsection{دسته‌بندی کلی بر اساس حسگر و نوع نما}
ادبیات را می‌توان بر اساس منبع هندسه به چهار گروه تقسیم کرد:
\begin{enumerate}
 \item \textbf{چندنما و چنددوربینه:} روش‌های استریو و چندنمایی با تطبیق نقاط متناظر و پارامترهای کالیبراسیون، عمق متریک را از چند تصویر استخراج می‌کنند. \lr{MVSNet} نمونه‌ای از این دسته است؛ ماتریس کالیبراسیون دوربین و وضعیت دوربین در ساخت حجم هزینه مستقیماً استفاده می‌شوند~\cite{Yao2018MVSNet}. این روش‌ها دقیق‌اند، اما به چند نمای هم‌زمان و کالیبراسیون نیاز دارند.
 \item \textbf{لایدار یا عمق حسگر:} نقاط پراکندهٔ لایدار یا نقشهٔ عمق رنگی--عمق (\lr{RGB-D})\enfootnote{RGB-D (Red--Green--Blue--Depth)} با تصویر ترکیب می‌شوند. این گروه برای آموزش متریک مناسب است، اما میدان دید محدود، پراکندگی و هزینهٔ حسگر، استفادهٔ مستقل از تصویر قرمز--سبز--آبی (\lr{RGB})\enfootnote{Red--Green--Blue} را محدود می‌کند.
 \item \textbf{تک‌دوربینه با ویدئو یا استریو در آموزش:} فقط یک تصویر در آزمون داده می‌شود، ولی آموزش از زوج استریو یا فریم‌های زمانی و هندسهٔ بازسازی استفاده می‌کند. این دسته مسیر اصلی خودنظارتی را شکل داده است.
 \item \textbf{تک‌تصویری بنیادین:} شبکه از یک تصویر \lr{RGB} خروجی نسبی یا متریک می‌سازد و با آموزش چندمجموعه‌ای، تقطیر یا مدل‌های انتشاری به تعمیم صفرنمونه می‌رسد. این دسته، محور اصلی مرور حاضر است.
\end{enumerate}

\subsection{روش‌های کلاسیک و نظارت‌شدهٔ اولیه}
پیش از شبکه‌های عمیق، روش‌های \lr{Hoiem} و \lr{Make3D} از خطوط همگرا، افق، سطوح، اندازهٔ نسبی اشیا و ویژگی‌های دستی برای برآورد ساختار صحنه استفاده می‌کردند~\cite{Hoiem2005GeometricContext,Saxena2009Make3D}. این روش‌ها رابطهٔ میان نشانهٔ دیداری و عمق را روشن کردند، اما به فرض‌های صحنه و دامنهٔ آموزش حساس بودند. \lr{Eigen} و همکاران یادگیری چندمقیاسی عمق را با شبکهٔ عمیق مطرح کردند و \lr{Eigen--Fergus} آن را به پیش‌بینی چندوظیفه‌ای گسترش دادند~\cite{Eigen2014Depth,Eigen2015Predicting}. خروجی این مدل‌ها درون‌دامنه متریک بود، اما تغییر دوربین و محیط، مقیاس آموخته‌شده را ناپایدار می‌کرد.

\subsection{سیر خودنظارتی: از استریو تا ویدئو}
\lr{Godard} و همکاران با جایگزین‌کردن عمق مرجع با زوج استریو، تابع بازسازی تصویر و سازگاری چپ--راست را معرفی کردند~\cite{Godard2017Unsupervised}. اگر $d_L,d_R$ دیسپاریتی‌های دو نما باشند، قید سازگاری به‌صورت کلی $L_{lr}=\lVert d_L-\mathcal W(d_R)\rVert$ نوشته می‌شود. این ایده آموزش \lr{RGB} را ممکن کرد، ولی مقیاس آن به خط پایه و کالیبراسیون استریوی داده گره خورد.
\lr{Zhou} و همکاران عمق و حرکت دوربین را از ویدئوی بدون برچسب با بازسازی دیدگاه هم‌زمان آموختند~\cite{Zhou2017Unsupervised}. \lr{Zhan} و همکاران با بازسازی ویژگی‌های عمیق در کنار خطای پیکسلی، پایداری بازسازی و تخمین ژست را بهبود دادند~\cite{Zhan2018DeepFeatureReconstruction}. مطالعهٔ نیمه‌نظارتی \lr{Amiri} و همکاران نیز ترکیب لایدار و سازگاری چپ--راست را بررسی کرد~\cite{Amiri2019SemiSupervised}. \lr{Monodepth2} با ماسک‌کردن خودکار پیکسل‌های نامعتبر و انتخاب حداقل بازسازی، خطاهای پویایی و انسداد را کاهش داد~\cite{Godard2019Digging}.

این خط پژوهش نشان داد که از یک تصویر می‌توان مدل عمق ساخت، اما برای خروجی متریک یا باید خط پایه/حرکت و ماتریس کالیبراسیون دوربین در حلقهٔ بازسازی معلوم باشند، یا مقیاس از دادهٔ آموزش به‌طور ضمنی آموخته شود. هیچ‌کدام به‌تنهایی تضمین نمی‌کند که شبکه برای هر دوربین جدید، مقیاس را به‌درستی تغییر دهد.

\subsection{مدل‌های نسبی، معماری‌های ترنسفورمری و مدل‌های سبک}
\lr{MiDaS} با ترکیب مجموعه‌داده‌های ناهمگون و هزینهٔ تغییرناپذیر نسبت به مقیاس و انتقال، عمق نسبی را به شکل قابل توجهی تعمیم داد~\cite{Ranftl2019Robust}. \lr{DPT} با استفاده از ویژگی‌های چندمرحله‌ای \lr{ViT}، کیفیت مرزها و سازگاری سراسری را بهبود داد~\cite{Ranftl2021DPT}. \lr{MiDaS v3.1} همین ایده را با ستون‌فقرات‌های \lr{BEiT}، \lr{Swin}، \lr{SwinV2}، \lr{Next-ViT} و \lr{LeViT} در طیف دقت--سرعت گسترش داد؛ بهترین مدل آن بهبود گزارش‌شدهٔ ۲۸ درصدی داشت، ولی خروجی همچنان نسبی است~\cite{MiDaSV31}.
\lr{Lite-Mono} ترکیب \lr{CNN} و ترنسفورمر را با بلوک‌های \lr{CDC} و \lr{LGFI} برای خودنظارتی سبک ارائه کرد و با حدود ۸۰ درصد پارامتر کمتر از \lr{Monodepth2} رقابت‌پذیر ماند~\cite{LiteMono}. این کارها نشان می‌دهند که معماری قوی‌تر دقت نسبی را بالا می‌برد، اما ظرفیت بیشتر به‌تنهایی ابهام مقیاس ناشی از دوربین را حل نمی‌کند.

\subsection{مدل‌های بنیادین و تقطیر در مقیاس بزرگ}
\lr{Depth Anything} با استفاده از تصاویر مصنوعی برچسب‌دار، تصاویر واقعی بدون برچسب و معلم‌های قوی، یادگیری عمق نسبی را از محدودیت مجموعه‌داده‌های کوچک جدا کرد~\cite{Yang2024DepthAnything}. \lr{Depth Anything V2} سه تغییر کلیدی دارد: جایگزینی تصاویر واقعی برچسب‌دار با تصاویر مصنوعی، افزایش ظرفیت معلم و آموزش دانش‌آموز با واسطهٔ تصاویر واقعی شبه‌برچسب‌خورده~\cite{DepthAnythingV2}. در جزئیات گزارش‌شدهٔ مقاله، معلم ابتدا روی پنج مجموعهٔ مصنوعی با مجموع حدود ۵۹۵ هزار تصویر آموزش می‌بیند؛ سپس خروجی معلم برای هشت مجموعهٔ واقعی با مجموع حدود ۶۲ میلیون تصویر تولید و برای آموزش دانش‌آموزان استفاده می‌شود. بنابراین برچسب‌های واقعیِ پرهزینه، با شبه‌برچسب‌های معلم جایگزین می‌شوند و چند اندازهٔ دانش‌آموز از حدود ۲۵ میلیون تا ۱٫۳ میلیارد پارامتر ارائه می‌گردد. این فرایند، پیش‌آموزش عمومی مدل پایه است و با فاین‌تیون متریک پژوهش حاضر یکسان نیست.

اگر $z_T$ خروجی معلم و $z_S$ خروجی دانش‌آموز باشند، یک قید معمول تقطیر مانند $L_{dist}=\lVert \mathcal N(z_S)-\mathcal N(z_T)\rVert$ پیش‌بینی دانش‌آموز را با بازنمایی نرمال‌شدهٔ معلم هم‌راستا می‌کند. نرمال‌سازی، تفاوت مقیاس مجموعه‌داده‌ها را کاهش می‌دهد، ولی خودِ آن اطلاعات ماتریس کالیبراسیون دوربین هر تصویر را ایجاد نمی‌کند. در مدل پایه، تصویر پس از تغییر اندازه و برش به انکودر \lr{DINOv2} داده می‌شود؛ ویژگی‌های چندمرحله‌ای از طریق دیکودر \lr{DPT} به نگاشت عمق چگال بازسازی می‌شوند. در نسخهٔ متریک، نویسندگان همین مدل‌های نسبی را با برچسب‌های عمق متریک فاین‌تیون می‌کنند؛ این مسیر، مبنای انتخاب وزن اولیه در این پایان‌نامه است، نه اینکه به‌تنهایی مسئلهٔ دوربین‌متغیر را حل کند.

\lr{Distill Any Depth} نشان می‌دهد که نرمال‌سازی سراسری شبه‌برچسب‌های پرنویز می‌تواند تقطیر را محدود کند؛ این کار با تقطیر متقاطع محلی--سراسری و چندمعلمه کیفیت را افزایش می‌دهد~\cite{DistillAnyDepth}. این نتیجه برای فصل حاضر مهم است، زیرا بهبود تقطیر و آگاهی از دوربین دو مسئلهٔ متفاوت‌اند و نباید یکی به‌جای دیگری تفسیر شود.

\subsection{مدل‌های متریک و شرطی‌سازی دوربین}
\lr{CAM-Convs} پارامترهای کالیبراسیون را به کانال‌های مختصاتی و میدان دید در کانولوشن وارد کرد و افت انتقال میان دوربین‌ها را کاهش داد~\cite{Facil2019CAMConvs}. این نخستین شاهد مستقیم در فهرست حاضر است که شبکهٔ تک‌تصویری از دانستن دوربین سود می‌برد.

\lr{ZoeDepth} یک ستون فقرات نسبی را با سرهای متریک دامنه‌ای ترکیب می‌کند~\cite{Bhat2023ZoeDepth}. این راهبرد مقیاس را در دامنه‌های شناخته‌شده می‌آموزد، اما برای دوربین یا دامنهٔ خارج از تقسیم‌بندی، کالیبراسیون صریحی ندارد.
\lr{Metric3D} ابهام را با نگاشت ورودی‌ها به فضای دوربین متعارف و آموزش روی داده‌های متنوع کاهش می‌دهد~\cite{Yin2023Metric3D}. \lr{Metric3D v2} این ایده را با «تبدیل فضای دوربین متعارف» و آموزش بیش از ۱۶ میلیون تصویر از هزاران مدل دوربین برای پیش‌بینی هم‌زمان عمق متریک و نرمال سطح گسترش می‌دهد~\cite{Metric3Dv2}. مزیت آن تعمیم صفرنمونه است؛ بااین‌حال اثر خالص دسترسی شبکه به ماتریس کالیبراسیون دوربین در برابر اثر تبدیل، داده و وظیفهٔ نرمال به‌طور جداگانه سنجیده نشده است.

مدل عمق متریک صفرنمونه (\lr{DMD}) با شرط میدان دید، از پارامترسازی لگاریتمی عمق، تنوع مصنوعی میدان دید و مدل انتشار برای ترکیب صحنه‌های داخلی و بیرونی استفاده می‌کند~\cite{Saxena2023FOV}. این کار نشان می‌دهد حتی یک اسکالر \lr{FoV} می‌تواند ابهام مقیاس را کاهش دهد، اما هزینهٔ استنتاج انتشار بالاست.
\lr{UniDepth} بازنمایی دوربین را از تصویر پیش‌بینی و ویژگی‌های عمق را با آن شرطی می‌کند؛ خروجی شبه‌کروی و زیان ناوردایی هندسی، جداسازی دوربین و عمق را تقویت می‌کنند~\cite{Piccinelli2024UniDepth}. این روش به دوربین معلوم در زمان آزمون وابسته نیست، ولی چون هم‌زمان ماژول دوربین، داده و زیان جدید را تغییر می‌دهد، سهم علّی یک ورودی ماتریس کالیبراسیون دوربین معلوم را مشخص نمی‌کند.

مدل‌های انتشار \lr{Marigold} از پیش‌دانش \lr{Stable Diffusion} برای عمق افاین‌ناوردا استفاده می‌کنند~\cite{Marigold}. \lr{Marigold-DC} مسئلهٔ دیگری را هدف می‌گیرد: تکمیل عمق با راهنمایی اندازه‌گیری‌های پراکنده در زمان آزمون~\cite{MarigoldDC}. بنابراین این دو کار شاهد قدرت \lr{prior}های مولدند، اما مقایسهٔ مستقیم آن‌ها با مدل متریک تک‌تصویری بدون اندازه‌گیری اضافی باید با احتیاط انجام شود.

\subsection{مقاله‌های پایه برای مدل پیشنهادی}
سه جزء پایهٔ انتخاب‌شده در این پژوهش مکمل یکدیگرند: \lr{DINOv2} بازنمایی تصویری مقاوم می‌سازد؛ \lr{DPT} بازنمایی چندمقیاسی را به خروجی چگال تبدیل می‌کند؛ و \lr{Depth Anything V2} با آموزش مصنوعی و تقطیر گسترده، نقطهٔ شروعی با تعمیم بالا فراهم می‌آورد. نکتهٔ روش‌شناختی این ترکیب آن است که هیچ‌یک از این سه جزء، در نسخهٔ پایه، بردار ماتریس کالیبراسیون دوربین را به‌صورت ورودی درجه‌یک در انکودر دریافت نمی‌کند. ازاین‌رو، انتقال مدل بنیادین به عمق متریک دوربین‌متغیر، پرسشی مستقل از کیفیت بازنمایی بصری باقی می‌گذارد.

\section{شکاف تحقیقاتی}
\label{sec:ch2_gap}
با توجه به مرور بالا، شکاف‌ها به‌صورت شماره‌دار زیر صورت‌بندی می‌شوند. در این بخش فقط مسئلهٔ باقی‌مانده و اهمیت آن بیان می‌شود؛ پاسخ‌های اجرایی در بخش بعد آمده‌اند.

\begin{enumerate}
\item \textbf{نبود اتصال صریح میان هندسهٔ دوربین و مدل بنیادین نسبی.} \lr{DPT}، \lr{MiDaS} و \lr{Depth Anything V2} عمدتاً از $I$ استفاده می‌کنند. حتی در روش‌های خودنظارتی که ماتریس کالیبراسیون دوربین در بازسازی حضور دارد، شبکهٔ عمق لزوماً آن را نمی‌بیند. بنابراین تغییر $f_x,f_y,c_x,c_y$ می‌تواند به تغییر مقیاس واقعی نیاز داشته باشد، بی‌آنکه مسیر صریحی برای اعمال این تغییر وجود داشته باشد.
\item \textbf{فاصلهٔ میان عمق نسبیِ بسیار تعمیم‌پذیر و عمق متریکِ وابسته به دامنه.} \lr{MiDaS} و \lr{Depth Anything} در خروجی نسبی قوی‌اند؛ \lr{ZoeDepth} و مدل‌های متریک درون‌دامنه مقیاس را بازیابی می‌کنند؛ \lr{Metric3D}، \lr{DMD} و \lr{UniDepth} تعمیم را با تغییر هم‌زمان فضا، داده یا ماژول دوربین دنبال می‌کنند. هنوز روشن نیست آیا یک مدل بنیادین موجود با افزودن اطلاعات هندسی کم‌بعد می‌تواند در رژیم فاین‌تیون محدود، متریک و قابل انتقال شود.
\item \textbf{نبود آزمایش عامل‌محور برای سهم خالص ماتریس کالیبراسیون دوربین.} در \lr{CAM-Convs}، \lr{Metric3D}، \lr{DMD} و \lr{UniDepth}، معماری، داده، پیش‌پردازش یا هدف آموزش هم‌زمان تغییر می‌کند. بهبود نهایی بنابراین اثر مشترک چند تصمیم است و نمی‌گوید چه مقدار از آن مستقیماً از دریافت ماتریس کالیبراسیون دوربین آمده است.
\item \textbf{کمبود آزمون سازوکاری برای تشخیص مصرف واقعی ماتریس کالیبراسیون دوربین.} بهبود \lr{AbsRel} به‌تنهایی ممکن است از هم‌بستگی داده‌ای یا افزایش ظرفیت ناشی شود. ادبیات کمتر می‌سنجد که آیا با تغییر کنترل‌شدهٔ $f$، خروجی با قانون هندسی $Z\propto f$ حرکت می‌کند و آیا حذف یا تحریف ماتریس کالیبراسیون دوربین دقت را به‌صورت قابل پیش‌بینی کاهش می‌دهد.
\item \textbf{آمیختگی ارزیابی نسبی و متریک.} تراز مقیاس در معیارهای نسبی می‌تواند خطایی را پنهان کند که برای کاربردهای فیزیکی مهم است. بدون گزارش هم‌زمان ارزیابی بدون تراز و ارزیابی تراز‌شده، نقش ماتریس کالیبراسیون دوربین در مقیاس مطلق قابل تشخیص نیست.
\item \textbf{خطرهای بازتولیدپذیری در زنجیرهٔ انتقال مدل.} نشت تقسیم‌بندی، ناسازگاری عمق مستقیم و دیسپاریتی، تغییر ماتریس کالیبراسیون دوربین در برش تصویر، یا ناهم‌خوانی سر متریک می‌تواند نتیجه‌ای ظاهراً معتبر بسازد و نسبت‌دادن اثر به هندسه را مخدوش کند. ادبیات معمولاً این کنترل‌های زیرساختی را به‌صورت یک آزمایش مستقل گزارش نمی‌کند.
\end{enumerate}

\section{نوآوری‌های تحقیق}
\label{sec:ch2_contributions}
نوآوری‌های زیر به‌ترتیب با شماره‌های بخش پیشین متناظرند و جزئیات فرمولی و پیاده‌سازی آن‌ها در فصل سوم می‌آید:

\begin{enumerate}
\item برای شکاف ۱، پارامترهای نرمال‌شدهٔ ماتریس کالیبراسیون دوربین به یک نمایش هندسی مستقل تبدیل و به جریان توکن‌های انکودر \lr{Depth Anything V2} افزوده شدند تا مدل تصویر و هندسهٔ تصویربرداری را هم‌زمان ببیند.
\item برای شکاف ۲، همان ستون فقرات بنیادین در دو حالت نسبی و متریک بررسی و با فاین‌تیون کنترل‌شده به مسئلهٔ عمق مطلق متصل شد؛ بدین ترتیب مقایسهٔ تعمیم بازنمایی پایه با بازیابی مقیاس ممکن شد.
\item برای شکاف ۳، مدل پیشنهادی و مدل کنترل با داده، تفکیک، ابرپارامتر و بذر یکسان به‌صورت زوجی آموزش داده شدند و دسترسی به ماتریس کالیبراسیون دوربین تنها عامل طراحی‌شدهٔ متفاوت بود.
\item برای شکاف ۴، آزمون پیشمایش ماتریس کالیبراسیون و آزمون حذف و تحریف ماتریس کالیبراسیون به‌عنوان سنجهٔ سازوکاری اجرا شدند تا پاسخ خروجی و افت دقت مستقیماً به مصرف اطلاعات دوربین نسبت داده شود.
\item برای شکاف ۵، معیارهای نسبیِ تراز‌شده و معیارهای متریکِ بدون تراز هم‌زمان گزارش شدند تا اثر حذف مقیاس در ارزیابی نسبی از نتیجه پنهان نشود.
\item برای شکاف ۶، پروتکل‌های بدون نشت، سازگاری فضای هدف، به‌روزرسانی هندسی ماتریس کالیبراسیون دوربین پس از برش و تطبیق سر خروجی در زنجیرهٔ آموزش و ارزیابی کنترل شدند.
\end{enumerate}

\section{جمع‌بندی فصل}
مرور تاریخی از روش‌های هندسی و چندنما، خودنظارتی استریو و ویدئویی، مدل‌های نسبی \lr{MiDaS}/\lr{DPT}، مدل‌های بنیادین \lr{Depth Anything} و روش‌های متریک \lr{CAM-Convs}، \lr{Metric3D}، \lr{DMD} و \lr{UniDepth} نشان داد که آگاهی از دوربین یک ایدهٔ تثبیت‌شده اما هنوز از نظر انتساب علّی و انتقال به مدل‌های بنیادین حل‌نشده است. فصل سوم اکنون فقط صورت‌بندی و اجرای روش پیشنهادی، انتخاب نمایش ماتریس کالیبراسیون دوربین، آموزش زوج کنترل و آزمایش‌های طراحی‌شده را شرح می‌دهد.
